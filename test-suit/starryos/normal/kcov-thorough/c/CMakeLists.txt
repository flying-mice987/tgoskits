cmake_minimum_required(VERSION 3.20)
project(test-kcov-thorough C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-kcov-thorough src/main.c)
target_include_directories(test-kcov-thorough PRIVATE src)
target_compile_options(test-kcov-thorough PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test-kcov-thorough PRIVATE -lpthread)
install(TARGETS test-kcov-thorough RUNTIME DESTINATION usr/bin)
