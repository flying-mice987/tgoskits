cmake_minimum_required(VERSION 3.20)
project(test-kcov C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-kcov src/main.c)
target_include_directories(test-kcov PRIVATE src)
target_compile_options(test-kcov PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-kcov RUNTIME DESTINATION usr/bin)
